\begin{abstract}
For the $n$ parallelograms obtained by matching the equal subintervals of the
two horizontal edges of the unit square, we prove that the minimum area of
their union is of order $\frac1{\log n}$, as conjectured by Kaczynski.
\end{abstract}
\maketitle
\pagestyle{plain}
\raggedbottom
\widowpenalty=10000
\clubpenalty=10000

Divide the top and bottom edges of the unit square into $n$ equal intervals,
match them by a permutation, and join matched intervals by parallelograms.
Their areas add to $1$; how small can their union be? Write $\alpha_n$ for
the minimum. The assertion $\alpha_n\asymp\frac1{\log n}$ is
\emph{KC2}.\footnote{KC2 abbreviates ``Kaczynski's Conjecture 2,'' following
its numbering in Kuklinski~\cite[\S\,3]{Kuklinski}.}

\begin{theorem}[KC2]\label{thm:kc2}
There is an absolute constant $C$ such that, for every $n\ge2$,
\[
 \frac1{H_n}\le\alpha_n\le\frac C{\log n},
 \qquad H_n=\sum_{j=1}^n\frac1j.
\]
\end{theorem}

A strip from $I$ on the bottom edge to $J$ on the top edge is encoded by
the endpoint square $I\times J$. Its section at height $t$ is
$\pi_t(I\times J)$, where $\pi_t(x,y)=(1-t)x+ty$.
Nesting suitable patterns of endpoint squares gives a single compact set. For compact $S\subset\R^2$, put
$S_\delta=S+[-\delta,\delta]^2$ and
\[
 \Fav(S)=\int_0^\pi|p_\theta(S)|\,d\theta,
 \qquad p_\theta(x,y)=x\cos\theta+y\sin\theta.
\]
We write $p_x,p_y$ for the coordinate projections, $|\cdot|$ for length or
area, and $\mathcal H^1$ for Hausdorff length.

\begin{theorem}\label{thm:favard}
There is a compact set $S\subset[0,1]^2$ such that
\[
 p_x(S)=p_y(S)=[0,1],\qquad \mathcal H^1(S)=1,
\]
and, for absolute constants $c,C>0$,
\[
 \frac c{\log(1/\delta)}\le\Fav(S_\delta)
 \le\frac C{\log(1/\delta)}\qquad(0<\delta<\frac12).
\]
Its length measure $\mu=\mathcal H^1|_S$ has uniform coordinate marginals:
$\mu(B\times[0,1])=\mu([0,1]\times B)=|B|$ for every Borel $B\subset[0,1]$.
\end{theorem}

Chang, Shmerkin and Suomala~\cite{CSS} obtain sharp Favard decay for random
Cantor sets. Cs\"ornyei and Suomala~\cite{CS} characterize the measures which
occur as one projection of a purely unrectifiable length measure. Here both
coordinate marginals are uniform, and the length is as small as a unit
projection permits. Nested permutation patterns on a fixed $L$-adic grid
have Favard length $\gtrsim_L\frac{\log j}{j}$ at scale $L^{-j}$
\cite[Theorem~A.1]{CSS}. We allow unequal widths during compression and
introduce the final equal grid only by rounding. Throughout,
logarithms are natural and $c,C$ denote positive absolute constants,
allowed to change.

\section{The overlap bound}

\begin{claim}\label{cl:lower}
Every union of $n$ permutation strips has area at least $\frac1{H_n}$.
\end{claim}
\begin{proof}
Let $Z$ count the strips covering a point, and let $A$ be their union area.
Since $\int Z=1$, Cauchy--Schwarz gives $1\le A\int Z^2$.
Two crossing strips whose endpoint index gaps are $p,q\ge1$ have signed
center displacement $\frac{p-(p+q)t}{n}$. Their intersection area is therefore
\[
 \frac1n\int_0^1\bigl(1-|p-(p+q)t|\bigr)_+\,dt
 =\frac1{n(p+q)}
 \le\frac1{4n}\left(\frac1p+\frac1q\right).
\]
Noncrossing pairs have no positive-area intersection. On either edge there
are $n-d$ pairs of index gap $d$. Expanding $Z^2$ counts each pair twice,
so
\[
 \int Z^2\le1+\frac1n\sum_{d=1}^{n-1}\frac{n-d}{d}=H_n.\qedhere
\]
\end{proof}

\section{The shuffle}

A \emph{pattern} is a union $P=\bigcup_{i=1}^N(I_i\times J_i)$, where each
family of closed intervals partitions $[0,1]$ up to endpoints, and
$|I_i|=|J_i|=w_i>0$. Write $N(P)=N$. For compact $E\subset\R^2$, define
\[
 \A(E)=\int_0^1|\pi_t(E)|\,dt.
\]
For a pattern, $\sum_iw_i=1$ and $\A(P)$ is the area of its strip union.
Each strip keeps its width $w_i$ at every height; only the overlaps change.

Fix a pattern $P$, and put $E_t=\pi_t(P)$, $\ell(t)=|E_t|$ and
$L(P)=\int_0^{1/2}\ell(t)\,dt$. Choose $0<w<1$ and partition a unit interval
into $m=\lceil\frac1w\rceil$ pieces of width $w$, with the last possibly
shorter. Make red and blue copies of this partition. On intervals of length
$2$, arrange their labels as follows:
\begin{equation}\label{eq:orders}
 \begin{array}{rl}
 \text{bottom:}&R_1,B_1,R_2,B_2,\ldots,R_m,B_m,\\
 \text{top:}&B_1,B_2,\ldots,B_m,R_1,R_2,\ldots,R_m.
 \end{array}
\end{equation}
Equal labels determine endpoint squares. Put a homothetic copy of $P$
inside each square, then divide both coordinates by $2$. Call the resulting
pattern $P_w$. It has $MN(P)$ squares, where $M=2m$. On each edge, the
endpoint intervals still partition $[0,1]$. Since the copy ratios sum to
$1$, subadditivity gives $|\pi(P_w)|\le|\pi(P)|$ for every linear projection
$\pi$.

Within either color, the enclosing squares occur in the same order on
both axes, so their $\pi_t$-projections have disjoint interiors. Each color
contributes exactly $\frac12\ell(t)$. Thus $\ell(t)-|\pi_t(P_w)|$ is the
overlap of the two colors.

Before scaling, a full-width blue copy starts $w$ to the right of its red
partner at the bottom and one unit to its left at the top. Dividing the
projected displacement $(1-t)w-t$ by the common width gives
$d_w(t)=(1-t)-\frac tw$. The full-width pieces occupy at least half the
unit interval, since $w\lfloor\frac1w\rfloor\ge\frac12$.
Their red copies have total side length at least $\frac14$ after scaling.
The red projections have disjoint interiors, so adding the paired overlaps
gives
\begin{equation}\label{eq:overlap}
 \ell(t)-|\pi_t(P_w)|\ge\frac14
       |E_t\cap(E_t+d_w(t))|.
\end{equation}

For measurable $E\subset[0,1]$, Fubini's theorem gives the overlap identity
\cite{Tao}
\begin{equation}\label{eq:convolution}
 \int_\R|E\cap(E+s)|\,ds=|E|^2,
 \qquad \int_0^1|E\cap(E-s)|\,ds=\frac12|E|^2.
\end{equation}
The first integral counts pairs in $E\times E$; the second follows by
symmetry.

\Needspace{9\baselineskip}
Choose one width $w=U^2$ for the whole pattern, with $U$ uniform on $(0,1)$.
To see why this law works, fix $0<t\le\frac12$. A displacement $d_w(t)=-s$,
where $0\le s\le1$, requires $w=\frac{t}{1-t+s}\asymp t$: small heights
need small widths. The corresponding change of variables is
\[
 U=\sqrt{\frac{t}{1-t+s}},\qquad
 |dU|=\frac{\sqrt t}{2(1-t+s)^{3/2}}\,ds\gtrsim\sqrt t\,ds.
\]
These widths lie in $(0,1]$. Equations~\eqref{eq:overlap}--\eqref{eq:convolution}
therefore give $\E_w[\ell(t)-|\pi_t(P_w)|]\gtrsim\sqrt t\,\ell(t)^2$.
Integrating in $t$ and applying weighted Cauchy--Schwarz, with
$\int_0^{1/2}\frac{dt}{\sqrt t}=\sqrt2$, gives
\begin{equation}\label{eq:sector}
 \E_w[\A(P)-\A(P_w)]\gtrsim
 \int_0^{1/2}\sqrt t\,\ell(t)^2\,dt
 \gtrsim L(P)^2.
\end{equation}
Uniform width sampling would instead leave the weight $t$; its reciprocal
is not integrable at zero, so this Cauchy--Schwarz step would fail.

\section{Small patterns and rounding}

\begin{claim}\label{cl:small}
For every integer $k\ge1$, there is a pattern $P$ such that
\[
 \A(P)\le\frac Ck,\qquad \log N(P)\le Ck.
\]
\end{claim}
\begin{proof}
Transposing $P$ exchanges $t$ and $1-t$ and preserves $\A(P)$ and $N(P)$.
Do this when necessary so that $L(P)\ge\frac12\A(P)$, then apply the shuffle.
Writing $P'$ for its result, \eqref{eq:sector} gives
\begin{equation}\label{eq:step}
 \E[\A(P)-\A(P')]\ge c\A(P)^2,
 \qquad
 \E\log\frac{N(P')}{N(P)}\le D,
\end{equation}
where $0<c<1$ is absolute and $D=\log4+2$. Indeed,
$M=2\lceil\frac1w\rceil\le\frac4{U^2}$ and
$\int_0^1\log\frac1u\,du=1$. We track logarithms because successive copy
counts multiply.

Start with the unit square. At each step, transpose if necessary and sample
a fresh independent $U$. Write $A_k,N_k$ for the area and number of squares
after $k$ steps, and $a_k=\E A_k$. The conditional form of
\eqref{eq:step} and Jensen's inequality give
\[
 a_{k+1}\le a_k-ca_k^2,\qquad
 \E\log N_k\le Dk.
\]
Since $a_0=1$ and $0<a_k\le1$, taking reciprocals gives
$\frac1{a_{k+1}}\ge\frac1{a_k}+c$, hence $a_k\le\frac1{1+ck}$.
To obtain both bounds for one pattern, use a single nonnegative sum:
\[
 \E\left[kA_k+\frac{\log N_k}{k}\right]
 \le\frac{k}{1+ck}+D\le c^{-1}+D.
\]
Some realization has this sum at most $c^{-1}+D$, so both summands are
bounded by the same constant. This proves the claim.
\end{proof}

\begin{claim}\label{cl:rounding}
For every pattern $P$ and every $n\ge2$,
\[
 \alpha_n\le\A(P)+\frac{2N(P)}n.
\]
\end{claim}
\begin{proof}
Each side of $I_i\times J_i$ contains at least $nw_i-2$ whole intervals
of the $n$-grid: only the two boundary intervals can be partly lost.
Pair as many as possible within each square. At least
$\sum_i(nw_i-2)=n-2N(P)$ pairs are matched, and their strips lie in the union
encoded by $P$. The numbers of unused intervals on the two edges agree;
match these arbitrarily. The at most $2N(P)$ added strips have total area
at most $\frac{2N(P)}n$.
\end{proof}

To finish Theorem~\ref{thm:kc2}, take
$k=\lfloor\frac{\log n}{2C}\rfloor$ for sufficiently large $n$, where $C$
is the constant in Claim~\ref{cl:small}. Then $N(P)\le\sqrt n$, so
\[
 \alpha_n\le\frac Ck+\frac2{\sqrt n}\lesssim\frac1{\log n}.
\]
Increasing the constant covers smaller $n$.
Claim~\ref{cl:lower} gives the lower bound.

\section{A compact set of length one}

The finite patterns already prove KC2. For the compact realization, use
$t=\frac{\sin\theta}{\cos\theta+\sin\theta}$ to write
\begin{equation}\label{eq:angular}
 \A(E)=\int_0^{\pi/2}\frac{|p_\theta(E)|}{(\cos\theta+\sin\theta)^3}\,d\theta.
\end{equation}
The weight lies between $\frac1{2\sqrt2}$ and $1$.

For patterns $P=\bigcup_i(v_i+w_i[0,1]^2)$ and $Q$, define insertion by
$P\star Q=\bigcup_i(v_i+w_iQ)$. It is again a pattern, with $N(P)N(Q)$
squares. Containment and $\sum_iw_i=1$ give, for every linear projection,
\begin{equation}\label{eq:insertion}
 |\pi(P\star Q)|\le\min\{|\pi(P)|,|\pi(Q)|\}.
\end{equation}
Reflection $R(x,y)=(x,1-y)$ exchanges the two angular quadrants:
$|p_\theta(RP)|=|p_{\pi-\theta}(P)|$. For $Q=P\star R(P)$, use the first
factor in \eqref{eq:insertion} on the first quadrant and the second factor
on the other. Equation~\eqref{eq:angular} gives
\[
 \Fav(Q)\le4\sqrt2\,\A(P),\qquad N(Q)=N(P)^2.
\]
Claim~\ref{cl:small}, with $k$ of order $2^j$, therefore supplies patterns
$P_j$ satisfying
\begin{equation}\label{eq:blocks}
 \Fav(P_j)\le2^{-j},\qquad \log N(P_j)\le C2^j.
\end{equation}

To obtain length $1$, the square covers must be grouped more efficiently:
their total diameter is always $\sqrt2$. For even $m$, let $B_m$ be the
equal-grid pattern given by the cyclic half-shift
$i\mapsto i+\frac m2\pmod m$, with $0\le i<m$. This pattern lies in the disk
$D_m$ centered at $(\frac12,\frac12)$ with diameter
\begin{equation}\label{eq:diameter}
 d_m=\sqrt{1+\frac4{m^2}}\longrightarrow1.
\end{equation}
In the upper-left half, $0\le x\le\frac12\le y\le1$ and
$|y-x-\frac12|\le\frac1m$. Checking the vertices of this polygon gives
maximum squared distance $\frac14+\frac1{m^2}$ from the center, attained at
$(0,\frac12+\frac1m)$. The other half is its central reflection.

Set $Q_j=B_{2j}\star P_j$, $T_0=[0,1]^2$, and
$T_j=T_{j-1}\star Q_j$. These patterns are nested, and
\begin{equation}\label{eq:nested}
 \Fav(T_j)\le2^{-j},\qquad
 \log N(T_j)\le\sum_{i=1}^j\bigl(\log(2i)+C2^i\bigr)\le B2^j
\end{equation}
for an absolute $B>0$. The factor $B_{2j}$ makes every child square at most
half as wide as its parent, so the largest side in $T_j$ is at most $2^{-j}$.
Let $S=\bigcap_jT_j$.

For each square $v_i+w_i[0,1]^2$ of $T_j$, the disk
$v_i+w_iD_{2(j+1)}$ covers all its descendants in $S$. These disks have
diameters tending uniformly to zero, and their total diameter is
\[
 \sum_iw_i\,d_{2(j+1)}=d_{2(j+1)}\longrightarrow1.
\]
Thus $\mathcal H^1(S)\le1$. Both coordinate projections of $S$ are $[0,1]$:
for each fixed coordinate, the corresponding fibers of the compact sets
$T_j$ are nonempty and nested. Projection cannot increase Hausdorff length,
so $\mathcal H^1(S)\ge1$ as well.

Length $1$ now forces the marginals. Put $\mu=\mathcal H^1|_S$. For either
coordinate projection $p$ and Borel $B\subset[0,1]$, the set
$S\cap p^{-1}(B)$ projects onto $B$, whence $\mu(p^{-1}(B))\ge|B|$.
The same inequality for the complement, together with total mass $1$,
forces equality.

To check the decay between construction scales, note that thickening a
union of $N$ squares by $[-\delta,\delta]^2$ increases its Favard length
by at most $8N\delta$: each projected interval gains
$2\delta(|\cos\theta|+|\sin\theta|)$. From \eqref{eq:nested},
\[
 \Fav(S_\delta)\le2^{-j}+8e^{B2^j}\delta.
\]
For sufficiently small $\delta$, choose the largest $j\ge1$ with
$e^{B2^j}\le\delta^{-1/2}$. Then $2^j\asymp\log\frac1\delta$ and
\[
 \Fav(S_\delta)\le\frac C{\log(1/\delta)}+8\sqrt\delta
 \le\frac C{\log(1/\delta)}.
\]
Increase $C$ to cover the remaining $0<\delta<\frac12$.

\section{The matching lower bound}

Let a compact $K\subset[0,1]^2$ carry a probability measure $\nu$ with
uniform coordinate marginals. Divide $[0,1]$ into $n$ equal intervals $I_i$,
half-open except for the last. The matrix
$a_{ij}=n\nu(I_i\times I_j)$ has every row and column sum equal to $1$.
Any $r$ rows have mass $r$, so their positive entries meet at least $r$
columns. Hall's theorem~\cite{Hall} gives a permutation $\sigma$ of positive
entries. Each selected cell meets $K$; closing the cells therefore gives
\[
 P_\sigma:=\bigcup_i\overline I_i\times\overline I_{\sigma(i)}
 \subset K_{1/n}.
\]
For $0<\delta<\frac12$, take $n=\lceil\frac1\delta\rceil$.
Claim~\ref{cl:lower} and \eqref{eq:angular} imply
\[
 \Fav(K_\delta)\ge\A(K_\delta)\ge\A(P_\sigma)
 \ge\frac1{H_n}\gtrsim\frac1{\log(1/\delta)}.
\]
Apply this to $K=S$ to complete Theorem~\ref{thm:favard}.
In particular, $\Fav(S)=0$ and $\mathcal H^1(S)=1$; the Besicovitch
projection theorem implies that $S$ is purely unrectifiable
(see \cite[\S1.1]{CSS}).

The upper estimates in the two theorems are equivalent. Matching at
$\delta=\frac1n$ gives KC2 for $n\ge3$, and $n=2$ is absorbed into the
constant. Conversely, KC2
supplies equal-grid patterns with $\A(P_j)\le2^{-j}$ and
$\log N(P_j)\le C2^j$ by taking $n_j=\lceil e^{C_*2^j}\rceil$.
Reflection, insertion, and the disk covers above then give
Theorem~\ref{thm:favard}.
